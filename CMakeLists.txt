cmake_minimum_required(VERSION 3.20)
project(SAFSYNCore VERSION 0.1.0 LANGUAGES CXX)

option(SAFSYN_BUILD_TESTS "Build the automated test suite" ON)
option(SAFSYN_BUILD_WINMM "Build the experimental Windows DLL shell" OFF)
option(SAFSYN_BUILD_BENCHMARKS "Build deterministic engine microbenchmarks" OFF)

add_library(safsyn STATIC
	SAFSYN/core.cpp
	SAFSYN/cohort_engine.cpp
	SAFSYN/engine.cpp
	SAFSYN/mastering.cpp
	SAFSYN/phase.cpp
	SAFSYN/playback.cpp
	SAFSYN/smf.cpp
	SAFSYN/smf_renderer.cpp
	SAFSYN/wav_writer.cpp
)
add_library(SAFSYN::safsyn ALIAS safsyn)
find_package(Threads REQUIRED)
target_link_libraries(safsyn PRIVATE Threads::Threads)
target_include_directories(safsyn PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/SAFSYN")
set(SAFSYN_UTILITY_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/utility/include")
if(NOT EXISTS "${SAFSYN_UTILITY_INCLUDE}/long_uint.h")
	message(FATAL_ERROR "Missing utility submodule; run git submodule update --init --recursive")
endif()

if(SAFSYN_BUILD_BENCHMARKS)
	add_executable(safsyn-engine-benchmark benchmarks/engine_benchmark.cpp)
	target_link_libraries(safsyn-engine-benchmark PRIVATE SAFSYN::safsyn)
	target_compile_features(safsyn-engine-benchmark PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(safsyn-engine-benchmark PRIVATE /EHsc)
	endif()
endif()
target_include_directories(safsyn PRIVATE "${SAFSYN_UTILITY_INCLUDE}")
target_compile_features(safsyn PUBLIC cxx_std_20)
target_compile_features(safsyn PRIVATE cxx_std_23)
set_target_properties(safsyn PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
	target_compile_options(safsyn PRIVATE /W4 /permissive- /EHsc)
	target_compile_definitions(safsyn PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
	target_compile_options(safsyn PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(safsyn-render tools/safsyn_render.cpp)
target_link_libraries(safsyn-render PRIVATE SAFSYN::safsyn)
target_compile_features(safsyn-render PRIVATE cxx_std_20)
if(MSVC)
	target_compile_options(safsyn-render PRIVATE /EHsc)
endif()

if(WIN32)
	add_library(safsyn-windows STATIC SAFSYN/windows_synth.cpp)
	target_link_libraries(safsyn-windows PUBLIC SAFSYN::safsyn PRIVATE ole32 uuid winmm avrt)
	target_compile_features(safsyn-windows PUBLIC cxx_std_20)
	target_compile_definitions(safsyn-windows PRIVATE UNICODE _UNICODE)
	add_executable(safsyn-synth WIN32 tools/safsyn_synth.cpp)
	target_link_libraries(safsyn-synth PRIVATE safsyn-windows comctl32 comdlg32 shell32)
	target_compile_definitions(safsyn-synth PRIVATE UNICODE _UNICODE NOMINMAX)
	add_executable(safsyn-play tools/safsyn_play.cpp)
	target_link_libraries(safsyn-play PRIVATE safsyn-windows)
	install(TARGETS safsyn-synth safsyn-play RUNTIME DESTINATION . COMPONENT Synth)
	install(FILES README.md LICENSE.txt DESTINATION . COMPONENT Synth)
	install(DIRECTORY docs/ DESTINATION docs COMPONENT Synth FILES_MATCHING PATTERN "*.md")
	if(MSVC)
		foreach(target safsyn-windows safsyn-synth safsyn-play)
			target_compile_options(${target} PRIVATE /EHsc /W4 /permissive-)
		endforeach()
		set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
		set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Synth)
		include(InstallRequiredSystemLibraries)
	endif()

	add_executable(safsyn-render-gui WIN32 tools/safsyn_render_gui.cpp)
	target_link_libraries(safsyn-render-gui PRIVATE SAFSYN::safsyn
		comctl32 comdlg32 shell32)
	target_compile_features(safsyn-render-gui PRIVATE cxx_std_20)
	target_compile_definitions(safsyn-render-gui PRIVATE UNICODE _UNICODE NOMINMAX)
	if(MSVC)
		target_compile_options(safsyn-render-gui PRIVATE /EHsc /W4 /permissive-)
	endif()
endif()

if(WIN32 AND SAFSYN_BUILD_WINMM)
	add_library(safsyn-winmm MODULE SAFSYN/dllmain.cpp)
	target_link_libraries(safsyn-winmm PRIVATE SAFSYN::safsyn)
	target_compile_features(safsyn-winmm PRIVATE cxx_std_20)
endif()

	target_link_libraries(safsyn-playback-tests PRIVATE SAFSYN::safsyn Threads::Threads)

		target_compile_options(safsyn-playback-tests PRIVATE /EHsc /W4)
	endif()

	target_compile_features(safsyn-tests PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(safsyn-tests PRIVATE /EHsc)
	endif()
	add_test(NAME safsyn-core COMMAND safsyn-tests "${CMAKE_CURRENT_BINARY_DIR}/test-data")
	add_executable(safsyn-phase-tests tests/phase_tests.cpp)
	target_link_libraries(safsyn-phase-tests PRIVATE SAFSYN::safsyn)
	target_compile_features(safsyn-phase-tests PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(safsyn-phase-tests PRIVATE /EHsc)
	endif()
	add_test(NAME safsyn-phase COMMAND safsyn-phase-tests)
	add_executable(safsyn-cohort-tests tests/cohort_tests.cpp)
	target_link_libraries(safsyn-cohort-tests PRIVATE SAFSYN::safsyn)
	target_compile_features(safsyn-cohort-tests PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(safsyn-cohort-tests PRIVATE /EHsc)
	endif()
	add_test(NAME safsyn-cohorts COMMAND safsyn-cohort-tests)
	set_tests_properties(safsyn-cohorts PROPERTIES TIMEOUT 60)
	add_executable(safsyn-mastering-tests tests/mastering_tests.cpp)
	target_link_libraries(safsyn-mastering-tests PRIVATE SAFSYN::safsyn)
	target_compile_features(safsyn-mastering-tests PRIVATE cxx_std_20)
	if(MSVC)

	target_link_libraries(safsyn-smf-tests PRIVATE SAFSYN::safsyn)

		target_compile_options(safsyn-smf-tests PRIVATE /EHsc)
	endif()
	add_test(NAME safsyn-smf COMMAND safsyn-smf-tests
		"${CMAKE_CURRENT_BINARY_DIR}/smf-test-data")
	add_test(NAME safsyn-demo-render
		COMMAND safsyn-render --demo "${CMAKE_CURRENT_BINARY_DIR}/coherent-demo.wav")
	add_test(NAME safsyn-coherent-sha
		COMMAND "${CMAKE_COMMAND}"
			"-DRENDERER=$<TARGET_FILE:safsyn-render>"
			"-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/coherent-sha.wav"
			"-DEXPECTED=457810A95A5EE047A6BD24E035221D68B6691A3E923EDB0D3BB9EA4B694456F5"
			-P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_coherent_sha.cmake")
endif()
